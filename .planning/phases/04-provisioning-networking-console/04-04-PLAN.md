```python
async def run_create_qemu(ctx: dict, job_id: int) -> None:
    def _build(connector, payload):
        node = payload["node"]; vmid = int(payload["vmid"]); config = payload["config"]
        async def _dispatch() -> str:
            return await connector.create_qemu(node=node, vmid=vmid, **config)
        return _dispatch
    await _run_polled_job(ctx, job_id, fn_name="run_create_qemu",
        audit_action="vm.create", build_dispatch=_build,
        target_id_from_payload=lambda p: str(p["vmid"]))
```
